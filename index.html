async function fetchPubChemByCas() { const rawCas = document.getElementById('addCas').value.trim(); const inputName = document.getElementById('addName').value.trim(); const btn = document.getElementById('btnFetchCas'); if (!rawCas && !inputName) { alert('請先輸入 CAS Number 或 藥品英文名稱！'); return; } btn.innerText = '⏳ 檢索中...'; btn.disabled = true; let smiles = ''; let formula = ''; let mw = ''; let fetchedName = ''; const cleanCas = rawCas.replace(/-/g, '').trim(); // 策略 1：查詢 CID let cid = null; for (let queryStr of [rawCas, cleanCas]) { if (!queryStr) continue; try { const cidRes = await fetch(`https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/substance/name/${encodeURIComponent(queryStr)}/cids/JSON`); if (cidRes.ok) { const cidData = await cidRes.json(); if (cidData.IdentifierList && cidData.IdentifierList.CID && cidData.IdentifierList.CID.length > 0) { cid = cidData.IdentifierList.CID[0]; break; } } } catch (e) {} } // 策略 2：根據 CID 取得完整屬性（囊括所有可能的 SMILES 欄位名稱） if (cid) { try { const propUrl = `https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/cid/${cid}/property/Title,CanonicalSMILES,IsomericSMILES,ConnectivitySMILES,MolecularFormula,MolecularWeight/JSON`; const propRes = await fetch(propUrl); if (propRes.ok) { const propData = await propRes.json(); const props = propData.PropertyTable.Properties[0]; smiles = props.CanonicalSMILES || props.IsomericSMILES || props.ConnectivitySMILES || ''; formula = props.MolecularFormula || ''; mw = props.MolecularWeight ? String(props.MolecularWeight) : ''; fetchedName = props.Title || ''; } } catch (e) {} } // 策略 3（備援）：若 CID 未取得 SMILES，直接以名稱/CAS 撈取 if (!smiles) { const fallbackKeywords = [rawCas, cleanCas, inputName].filter(Boolean); for (let kw of fallbackKeywords) { try { const directRes = await fetch(`https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/${encodeURIComponent(kw)}/property/Title,CanonicalSMILES,IsomericSMILES,ConnectivitySMILES,MolecularFormula,MolecularWeight/JSON`); if (directRes.ok) { const directData = await directRes.json(); if (directData.PropertyTable && directData.PropertyTable.Properties && directData.PropertyTable.Properties.length > 0) { const p = directData.PropertyTable.Properties[0]; smiles = p.CanonicalSMILES || p.IsomericSMILES || p.ConnectivitySMILES || ''; formula = formula || p.MolecularFormula || ''; mw = mw || (p.MolecularWeight ? String(p.MolecularWeight) : ''); fetchedName = fetchedName || p.Title || ''; if (smiles) break; } } } catch (e) {} } } btn.innerText = '🔍 自動帶入'; btn.disabled = false; if (smiles || formula || mw) { if (smiles) { document.getElementById('addSmiles').value = smiles; previewStructure(); } if (formula) document.getElementById('addFormula').value = formula; if (mw) document.getElementById('addMw').value = mw; if (fetchedName && !document.getElementById('addName').value.trim()) { document.getElementById('addName').value = fetchedName; } alert(`✨ 成功帶入結構與分子數據！\n・SMILES: ${smiles}\n・分子式: ${formula}\n・分子量: ${mw} g/mol`); } else { alert('PubChem 未能辨識此 CAS 號碼，請確認號碼是否有誤，或手動填入 SMILES 結構式。'); } }
